Here are three production-grade Python scripts that download **real datasets**, train on **real network attack traffic**, and export **real pre-trained models** to ONNX with zero synthetic/dummy data:

1. **`tools/train_cicids2017_onnx.py`**: Downloads the real **CIC-IDS-2017 PortScan dataset** directly from Hugging Face, trains a deep neural network on the actual network flow telemetry, and exports `models/network_threat.onnx`.
2. **`tools/export_vision_yolo.py`**: Downloads the official pre-trained **YOLOv11n** weights (trained on the genuine 80-class MS COCO dataset for human intruders, vehicles, and weapons) and exports `models/vision_yolo.onnx`.
3. **`tools/export_log_nlp.py`**: Downloads the genuine pre-trained **Wazuh SecRoBERTa Security Log Classifier** (`pyToshka/wazuh-assist` trained on real security logs) from Hugging Face and exports `models/log_nlp.onnx`.

### Model 1: Real CIC-IDS-2017 Network Intrusion Model (`tools/train_cicids2017_onnx.py`)

This script downloads the actual labeled CSV from the Canadian Institute for Cybersecurity (CIC) mirror on Hugging Face (`c01dsnap/CIC-IDS2017`), extracts 32 standard flow features, trains on real port scan and benign traffic, and exports to `models/network_threat.onnx`.

#### File: `tools/train_cicids2017_onnx.py`
```python
#!/usr/bin/env python3
import os
import requests
import pandas as pd
import numpy as np
import torch
import torch.nn as nn
import torch.optim as optim
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import StandardScaler
from tqdm import tqdm

DATASET_URL = "https://huggingface.co/datasets/c01dsnap/CIC-IDS2017/resolve/main/Friday-WorkingHours-Afternoon-PortScan.pcap_ISCX.csv"
DATASET_CSV = "tools/cicids2017_portscan.csv"
MODEL_OUTPUT = "models/network_threat.onnx"

FEATURE_COLS = [
    "Destination Port", "Flow Duration", "Total Fwd Packets", "Total Backward Packets",
    "Total Length of Fwd Packets", "Total Length of Bwd Packets", "Fwd Packet Length Max",
    "Fwd Packet Length Min", "Fwd Packet Length Mean", "Fwd Packet Length Std",
    "Bwd Packet Length Max", "Bwd Packet Length Min", "Bwd Packet Length Mean",
    "Bwd Packet Length Std", "Flow Bytes/s", "Flow Packets/s", "Flow IAT Mean",
    "Flow IAT Std", "Flow IAT Max", "Flow IAT Min", "Fwd IAT Total",
    "Fwd IAT Mean", "Fwd IAT Std", "Fwd IAT Max", "Fwd IAT Min",
    "Bwd IAT Total", "Bwd IAT Mean", "Bwd IAT Std", "Bwd IAT Max",
    "Bwd IAT Min", "Fwd PSH Flags", "Bwd PSH Flags"
]

class NetworkThreatClassifier(nn.Module):
    def __init__(self, input_dim=32):
        super(NetworkThreatClassifier, self).__init__()
        self.net = nn.Sequential(
            nn.Linear(input_dim, 64),
            nn.BatchNorm1d(64),
            nn.ReLU(),
            nn.Dropout(0.2),
            nn.Linear(64, 32),
            nn.ReLU(),
            nn.Linear(32, 16),
            nn.ReLU(),
            nn.Linear(16, 2),
            nn.Softmax(dim=1)
        )

    def forward(self, x):
        return self.net(x)

def download_dataset():
    if os.path.exists(DATASET_CSV):
        print(f"[CIC-IDS2017] Using cached dataset: {DATASET_CSV}")
        return

    print(f"[CIC-IDS2017] Downloading real CIC-IDS2017 PortScan dataset (77 MB)...")
    resp = requests.get(DATASET_URL, stream=True)
    total = int(resp.headers.get('content-length', 0))
    with open(DATASET_CSV, 'wb') as f, tqdm(total=total, unit='B', unit_scale=True) as bar:
        for chunk in resp.iter_content(chunk_size=1024 * 1024):
            f.write(chunk)
            bar.update(len(chunk))
    print("[CIC-IDS2017] Download complete!")

def main():
    download_dataset()

    print("[CIC-IDS2017] Loading dataset with pandas...")
    df = pd.read_csv(DATASET_CSV, encoding='utf-8', low_memory=False)
    df.columns = df.columns.str.strip()

    # Clean missing and infinite values
    df.replace([np.inf, -np.inf], np.nan, inplace=True)
    df.dropna(subset=FEATURE_COLS + ['Label'], inplace=True)

    # Encode Real Labels: BENIGN = 0, PortScan / Attack = 1
    labels = (df['Label'] != 'BENIGN').astype(int).values
    features = df[FEATURE_COLS].astype(np.float32).values

    print(f"[CIC-IDS2017] Total Processed Samples: {len(features)} | Attacks: {labels.sum()} | Benign: {(labels == 0).sum()}")

    # Normalize features using real dataset statistics
    scaler = StandardScaler()
    features = scaler.fit_transform(features)

    # Split 80/20 train/test
    X_train, X_test, y_train, y_test = train_test_split(features, labels, test_size=0.2, random_state=42)

    X_train_t = torch.tensor(X_train, dtype=torch.float32)
    y_train_t = torch.tensor(y_train, dtype=torch.long)
    X_test_t = torch.tensor(X_test, dtype=torch.float32)
    y_test_t = torch.tensor(y_test, dtype=torch.long)

    # Train PyTorch Model
    model = NetworkThreatClassifier(input_dim=32)
    criterion = nn.CrossEntropyLoss()
    optimizer = optim.Adam(model.parameters(), lr=0.002)

    print("[CIC-IDS2017] Training Neural Network on real network telemetry flows...")
    batch_size = 2048
    dataset = torch.utils.data.TensorDataset(X_train_t, y_train_t)
    loader = torch.utils.data.DataLoader(dataset, batch_size=batch_size, shuffle=True)

    model.train()
    for epoch in range(5):
        epoch_loss = 0.0
        for batch_x, batch_y in loader:
            optimizer.zero_grad()
            out = model(batch_x)
            loss = criterion(out, batch_y)
            loss.backward()
            optimizer.step()
            epoch_loss += loss.item()
        print(f"Epoch [{epoch+1}/5] - CrossEntropy Loss: {epoch_loss/len(loader):.5f}")

    # Evaluate Model on Test Set
    model.eval()
    with torch.no_grad():
        test_preds = model(X_test_t)
        predicted_classes = torch.argmax(test_preds, dim=1).numpy()
        acc = (predicted_classes == y_test).mean()
        print(f"[CIC-IDS2017] Final Test Set Accuracy on Real Traffic: {acc * 100.0:.2f}%")

    # Export directly to ONNX
    dummy_input = torch.randn(1, 32, dtype=torch.float32)
    torch.onnx.export(
        model,
        dummy_input,
        MODEL_OUTPUT,
        export_params=True,
        opset_version=17,
        do_constant_folding=True,
        input_names=['input'],
        output_names=['scores'],
        dynamic_axes={'input': {0: 'batch_size'}, 'scores': {0: 'batch_size'}}
    )
    print(f"[SUCCESS] Exported real trained model to: {MODEL_OUTPUT} ({os.path.getsize(MODEL_OUTPUT)} bytes)")

if __name__ == "__main__":
    main()
```

### Model 2: Genuine YOLOv11n Vision Perimeter Model (`tools/export_vision_yolo.py`)

This script downloads the official, real pre-trained **YOLO11n weights** (trained on 118,000 real images in MS COCO for human intruders, weapons, backpacks, and vehicles) and exports it to a static `640x640` ONNX graph ready for `xinfer`.

#### File: `tools/export_vision_yolo.py`
```python
#!/usr/bin/env python3
import os
import shutil
from ultralytics import YOLO

def main():
    print("[Vision Perimeter] Loading official real pre-trained YOLO11n weights (MS COCO dataset)...")
    model = YOLO("yolo11n.pt") # Automatically downloads official weights from Ultralytics

    print("[Vision Perimeter] Exporting real vision model to ONNX...")
    exported_file = model.export(
        format="onnx",
        imgsz=640,
        opset=17,
        dynamic=False,
        simplify=True
    )

    dest_file = "models/vision_yolo.onnx"
    shutil.move(exported_file, dest_file)
    print(f"[SUCCESS] Real YOLOv11n model exported to: {dest_file} ({os.path.getsize(dest_file)} bytes)")

if __name__ == "__main__":
    main()
```

### Model 3: Real Security Log Classifier (`tools/export_log_nlp.py`)

This script downloads **Wazuh SecRoBERTa** (`pyToshka/wazuh-assist`), an open-source Transformer trained on real enterprise security and audit logs, and exports it directly to ONNX format.

#### File: `tools/export_log_nlp.py`
```python
#!/usr/bin/env python3
import os
import torch
from transformers import AutoTokenizer, AutoModelForSequenceClassification

MODEL_NAME = "pyToshka/wazuh-assist"
OUTPUT_ONNX = "models/log_nlp.onnx"

def main():
    print(f"[Log NLP] Downloading real pre-trained security log classifier: {MODEL_NAME} ...")
    tokenizer = AutoTokenizer.from_pretrained(MODEL_NAME)
    model = AutoModelForSequenceClassification.from_pretrained(MODEL_NAME)
    model.eval()

    # Real test security log line
    test_log = "Failed password for root from 192.168.1.250 port 54321 ssh2"
    inputs = tokenizer(test_log, return_tensors="pt", max_length=64, padding="max_length", truncation=True)

    print(f"[Log NLP] Verifying on real log: '{test_log}'")
    with torch.no_grad():
        outputs = model(**inputs)
        predicted_class = torch.argmax(outputs.logits, dim=-1).item()
        class_names = ["benign", "suspicious", "malicious"]
        print(f"[Log NLP] Real Model Prediction: {class_names[predicted_class]}")

    print("[Log NLP] Exporting Transformer model to ONNX format...")
    dummy_input_ids = inputs["input_ids"]
    dummy_mask = inputs["attention_mask"]

    class OnnxWrapper(torch.nn.Module):
        def __init__(self, base_model):
            super().__init__()
            self.base_model = base_model

        def forward(self, input_ids, attention_mask):
            out = self.base_model(input_ids=input_ids, attention_mask=attention_mask)
            return torch.softmax(out.logits, dim=-1)

    wrapped_model = OnnxWrapper(model)
    wrapped_model.eval()

    torch.onnx.export(
        wrapped_model,
        (dummy_input_ids, dummy_mask),
        OUTPUT_ONNX,
        export_params=True,
        opset_version=17,
        do_constant_folding=True,
        input_names=['input_ids', 'attention_mask'],
        output_names=['scores'],
        dynamic_axes={
            'input_ids': {0: 'batch_size'},
            'attention_mask': {0: 'batch_size'},
            'scores': {0: 'batch_size'}
        }
    )

    print(f"[SUCCESS] Real Log Security NLP model exported to: {OUTPUT_ONNX} ({os.path.getsize(OUTPUT_ONNX)} bytes)")

if __name__ == "__main__":
    main()
```

To understand why this is different from YOLO, it helps to distinguish between an **AI model architecture** and a **benchmark dataset**:

* **YOLO** is an **AI model family** developed by Ultralytics. They train and distribute official pre-trained checkpoint weights (`yolo11n.pt`, `yolov8n.onnx`) directly.
* **CIC-IDS-2017** is a **benchmark dataset** created by the Canadian Institute for Cybersecurity (University of New Brunswick). The institute publishes raw PCAP network captures and CSV flow telemetry, not an "official" neural network model.

---

### Do Pre-Trained Models for CIC-IDS-2017 Exist?

**Yes.** While the Canadian Institute for Cybersecurity does not publish official weights, independent researchers and security engineers have trained models on CIC-IDS-2017 and published the pre-trained weights on Hugging Face:

| Repository / Model on Hugging Face | Pre-Trained Weights Format | Architecture / Task |
| :--- | :---: | :--- |
| **`darkknight25/ddos_xgboost_onnx`** | **Direct `.onnx`** | Network flow DDoS & intrusion detector trained on the CIC dataset, exported to ONNX. |
| **`gehad-alaa-abaas/RNN-IDS-MODEL`** | PyTorch (`.pth`) | PyTorch LSTM trained on the 80 flow features of CIC-IDS-2017 (binary & multi-class). |
| **`mehddii/cicids2017-soc-classifier-v2`** | Scikit-Learn / LightGBM | Two-stage Isolation Forest + Classifier trained on CIC-IDS-2017. |

### How to Download a Pre-Trained CIC Flow Model (No Training Required)

If you do not want to run the training script from the previous step and prefer to download pre-trained weights directly, you can download **`darkknight25/ddos_xgboost_onnx`** or convert **`gehad-alaa-abaas/RNN-IDS-MODEL`** from Hugging Face.

Here is a Python script that downloads the pre-trained CIC-trained model directly from Hugging Face into `models/network_threat.onnx`:

#### File: `tools/download_pretrained_cic_model.py`
```python
#!/usr/bin/env python3
import os
import urllib.request
from huggingface_hub import hf_hub_download

OUTPUT_DIR = "models"
MODEL_DEST = os.path.join(OUTPUT_DIR, "network_threat.onnx")

def main():
    os.makedirs(OUTPUT_DIR, exist_ok=True)

    print("[Model Hub] Downloading pre-trained network flow model trained on the CIC dataset...")
    
    # Download pre-trained ONNX model from Hugging Face repository
    repo_id = "darkknight25/ddos_xgboost_onnx"
    filename = "ddos_detection_model.onnx"

    try:
        downloaded_path = hf_hub_download(repo_id=repo_id, filename=filename)
        
        # Copy to models/network_threat.onnx
        import shutil
        shutil.copy(downloaded_path, MODEL_DEST)
        
        print(f"[SUCCESS] Downloaded pre-trained CIC model to: {MODEL_DEST}")
        print(f"File size: {os.path.getsize(MODEL_DEST)} bytes")
        
    except Exception as e:
        print(f"[Fallback] Downloading directly via HTTPS: {e}")
        url = f"https://huggingface.co/{repo_id}/resolve/main/{filename}"
        urllib.request.urlretrieve(url, MODEL_DEST)
        print(f"[SUCCESS] Downloaded pre-trained CIC model via URL to: {MODEL_DEST}")

if __name__ == "__main__":
    main()
```

### Summary
1. **For Computer Vision:** YOLO has official Ultralytics pre-trained weights (`yolo11n.pt` / `yolov8n.onnx`).
2. **For Network Telemetry:** Pre-trained weights are hosted by the community on Hugging Face (such as `darkknight25/ddos_xgboost_onnx` or `gehad-alaa-abaas/RNN-IDS-MODEL`). 
3. **Training vs. Pre-Trained:** Training on the real CSV with `train_cicids2017_onnx.py` gives you an exact 32-feature input shape tailored to `blackbox`, whereas using pre-trained weights from Hugging Face allows you to test immediately without downloading the 77\,MB dataset.
