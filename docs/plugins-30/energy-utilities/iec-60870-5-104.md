# IEC 60870-5-104 High-Voltage Telecontrol Dissector (`libsentinel_plugin_iec104.so`)

The IEC 60870-5-104 (IEC 104) dissector inspects electric transmission and distribution telecontrol traffic communicating over TCP port **2404**. It parses Application Protocol Control Information (APCI) frames and Application Service Data Units (ASDU), intercepting unauthorized switchgear commands and grid sabotage attempts (e.g., **Industroyer / CrashOverride**).

## 2. In-Memory Attack Signature Trapping (`Iec104Dissector.cpp`)

```cpp
#include <sentinel/dissector_interface.hpp>
#include <arpa/inet.h>
#include <cstring>

namespace sentinel::plugins {

#pragma pack(push, 1)
struct ApciHeader {
    uint8_t start_byte; // Must be 0x68
    uint8_t apdu_length;
    uint8_t control_field[4];
};

struct AsduHeader {
    uint8_t  type_id;
    uint8_t  num_objects; // Bit 7: Sequence flag, Bits 0-6: Count
    uint16_t cot;         // Cause of Transmission
    uint16_t ca;          // Common Address of ASDU
};
#pragma pack(pop)

class Iec104Dissector final : public IDissectorPlugin {
public:
    std::string_view protocol_name() const noexcept override { return "IEC_60870_5_104"; }
    uint16_t default_port() const noexcept override { return 2404; }
    uint32_t dissector_id() const noexcept override { return 201; }

    DissectionResult dissect(std::span<const uint8_t> payload, uint64_t timestamp_ns) noexcept override {
        DissectionResult res{};
        if (payload.size() < sizeof(ApciHeader) + sizeof(AsduHeader)) {
            return res;
        }

        const auto* apci = reinterpret_cast<const ApciHeader*>(payload.data());
        if (apci->start_byte != 0x68) return res;

        res.is_protocol_match = true;

        // Check if I-Format frame (Bit 0 of Control Field 1 must be 0)
        if ((apci->control_field[0] & 0x01) == 0) {
            const auto* asdu = reinterpret_cast<const AsduHeader*>(payload.data() + sizeof(ApciHeader));

            // Trap Industroyer Command Types:
            // Type 45 = C_SC_NA_1 (Single Command)
            // Type 46 = C_DC_NA_1 (Double Command - Breaker Open/Close)
            if (asdu->type_id == 45 || asdu->type_id == 46) {
                // Parse 3-byte Information Object Address (IOA)
                size_t ioa_offset = sizeof(ApciHeader) + sizeof(AsduHeader);
                if (payload.size() >= ioa_offset + 3) {
                    uint32_t ioa = payload[ioa_offset] | 
                                  (payload[ioa_offset + 1] << 8) | 
                                  (payload[ioa_offset + 2] << 16);

                    // Restrict critical substation IOA ranges (e.g. Substation Breakers)
                    if (ioa >= 1000 && ioa <= 1100) {
                        res.is_anomaly_detected = true;
                        res.rule_id = 2401;
                        res.severity = 4; // Critical
                        std::strncpy(res.description, "Unauthorized High-Voltage Breaker Trip Command", 63);
                    }
                }
            }
        }
        return res;
    }
};

} // namespace sentinel::plugins
```
